// ======== 读取 data.json 并渲染 ======== fetch('data.json') .then(response => response.json()) .then(data => { // 1. 替换所有基础文字和图片 document.querySelectorAll('[data-pages]').forEach(el => { let key = el.getAttribute('data-pages'); if (data[key]) { if(el.tagName === 'IMG') el.src = data[key]; else if (el.tagName === 'DIV' && key === 'coverSub') el.innerHTML = data[key].replace(/\n/g, '
'); else if (el.classList.contains('sound-note')) el.innerHTML = data[key].replace(/\n/g, '
'); else el.innerText = data[key]; } }); // 2. 渲染业务目录 let indexHtml = ''; if(data.businesses) { data.businesses.forEach((biz, i) => { indexHtml += `
0${i+1}${biz.title}${biz.desc}
`; }); } indexHtml += `
*私人联系点击进入
`; document.getElementById('index-list-container').innerHTML = indexHtml; // 3. 渲染日记 if(data.diaries) { let diaryHtml = ''; data.diaries.slice(0, 100).forEach(d => { diaryHtml += `
${d.date}
${d.content}
`; }); document.getElementById('diary-list-container').innerHTML = diaryHtml; } // 4. 渲染须知目录 if(data.notices) { let noticeMainHtml = ''; data.notices.forEach((notice, i) => { noticeMainHtml += `
0${i+1} / ${notice.title.substring(0,2).toUpperCase()}
${notice.title}
${notice.subTitle}
`; }); document.getElementById('notice-main-container').innerHTML = noticeMainHtml; } // 全局变量存一下，给点开业务和须知用 window.siteData = data; // 颜色替换 if(data.themeColor) { document.documentElement.style.setProperty('--blue', '#' + data.themeColor); document.documentElement.style.setProperty('--glitch', '#' + data.themeColor); } }) .catch(err => console.log("数据加载失败", err)); // 下面的 openCategory、openWork、openNotice、switchTab 保持原样
